function Skip-CopyFile {
    param(
        [string]$src,
        [string]$dst,
        [int]$skipBytes
    )
    $fsIn = [System.IO.File]::OpenRead($src)
    $fsOut = [System.IO.File]::Create($dst)
    try {
        $fsIn.Seek($skipBytes, [System.IO.SeekOrigin]::Begin)
        $buf = New-Object byte[] 1MB
        while( $n = $fsIn.Read($buf,0,$buf.Length) ){
            $fsOut.Write($buf,0,$n)
        }
    }finally{
        $fsIn.Close()
        $fsOut.Close()
    }
}
